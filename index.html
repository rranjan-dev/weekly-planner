Setup Instructions (one-time, 5 min)

  1. Open Google Sheets → Create a blank sheet
  2. Name it "Weekly Planner Data" (optional)
  3. In Row 1, type: Key in A1, Data in B1, Updated in C1
  4. Go to Extensions → Apps Script
  5. Delete any code there, paste this:
function doGet(e) {
  var sh = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sh.getDataRange().getValues();
  var data = {};
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) try { data[rows[i][0]] = JSON.parse(rows[i][1]); } catch(e) {}
  }
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sh = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var keys = Object.keys(payload);
  for (var k = 0; k < keys.length; k++) {
    var key = keys[k], val = JSON.stringify(payload[key]);
    var rows = sh.getDataRange().getValues();
    var found = false;
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] === key) {
        sh.getRange(i+1,2).setValue(val);
        sh.getRange(i+1,3).setValue(new Date().toISOString());
        found = true; break;
      }
    }
    if (!found) sh.appendRow([key, val, new Date().toISOString()]);
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
